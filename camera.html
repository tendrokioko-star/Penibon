function drawFaceEffect(){ if(!facePoints)return; const leftEye=point(33); const rightEye=point(263); const nose=point(1); const leftCheek=point(205); const rightCheek=point(425); if(!leftEye||!rightEye||!nose)return; const dx=rightEye.x-leftEye.x; const dy=rightEye.y-leftEye.y; const eyeDistance=Math.hypot(dx,dy); const angle=Math.atan2(dy,dx); ctx.save(); ctx.textAlign="center"; ctx.textBaseline="middle"; if(effect==="glasses"){ const lensWidth=eyeDistance*.48; const lensHeight=eyeDistance*.34; const centerX=(leftEye.x+rightEye.x)/2; const centerY=(leftEye.y+rightEye.y)/2; ctx.translate(centerX,centerY); ctx.rotate(angle); // Sunglasses lenses ctx.fillStyle="rgba(12,18,30,.88)"; ctx.strokeStyle="#111"; ctx.lineWidth=Math.max(3,eyeDistance*.045); [-1,1].forEach(side=>{ ctx.beginPath(); ctx.roundRect( side*eyeDistance*.27-lensWidth/2, -lensHeight/2, lensWidth, lensHeight, lensHeight*.25 ); ctx.fill(); ctx.stroke(); }); // Bridge ctx.beginPath(); ctx.moveTo(-eyeDistance*.06,0); ctx.lineTo(eyeDistance*.06,0); ctx.stroke(); // Side arms ctx.beginPath(); ctx.moveTo(-eyeDistance*.51,0); ctx.lineTo(-eyeDistance*.68,-lensHeight*.12); ctx.moveTo(eyeDistance*.51,0); ctx.lineTo(eyeDistance*.68,-lensHeight*.12); ctx.stroke(); } if(effect==="hearts"){ ctx.font=(eyeDistance*.45)+"px Arial"; ctx.fillText("❤️",nose.x,nose.y-eyeDistance*1.5); ctx.fillText("💕",nose.x-eyeDistance,nose.y); } if(effect==="stars"){ ctx.font=(eyeDistance*.4)+"px Arial"; ctx.fillText("✨",leftEye.x-eyeDistance*.3,leftEye.y); ctx.fillText("✨",rightEye.x+eyeDistance*.3,rightEye.y); ctx.fillText("⭐",nose.x,nose.y-eyeDistance); } if(effect==="blush"&&leftCheek&&rightCheek){ ctx.fillStyle="rgba(255,75,135,.35)"; [leftCheek,rightCheek].forEach(p=>{ ctx.beginPath(); ctx.ellipse(p.x,p.y,eyeDistance*.24, eyeDistance*.13,angle,0,Math.PI*2); ctx.fill(); }); } if(effect==="freckles"&&leftCheek&&rightCheek){ ctx.fillStyle="#9a6045"; for(let side of [leftCheek,rightCheek]){ for(let i=0;i<7;i++){ ctx.beginPath(); ctx.arc( side.x+(Math.random()-.5)*eyeDistance*.35, side.y+(Math.random()-.5)*eyeDistance*.2, 2.5,0,Math.PI*2 ); ctx.fill(); } } } ctx.restore(); }
